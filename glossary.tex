\newglossaryentry{wakefield}
{ name={wakefield}, description={An electromagnetic field generated by a charged particle beam interacting with its surrounding structure} }

\newglossaryentry{impedance}
{ name={impedance}, description={A frequency-domain quantity describing the interaction between a beam and an accelerator component} }

\newglossaryentry{stl}
{ name={STL}, description={A file format used to represent the surface geometry of three-dimensional objects} }

\newglossaryentry{gdfidl}
{ name={GdfidL}, description={An open-source time-domain wakefield solver based on the \gls{fdtd} method, using a uniform Cartesian mesh and a co-moving window approach for short-range impedance simulations}, }

\newglossaryentry{pbci}
{ name={PBCI}, description={A closed-source time-domain wakefield solver based on the \gls{fit}, used for beam-coupling impedance simulations in particle accelerators}, }

\newglossaryentry{emclaw}
{ name={EMcLaW}, description={An open-source time-domain electromagnetic solver implementing a finite-volume Godunov scheme for Maxwell's equations}, }

\newglossaryentry{numpy}
{ name={NumPy}, description={An open-source Python library providing efficient multi-dimensional array operations and linear algebra routines for scientific computing}, }

\newglossaryentry{scipy}
{ name={SciPy}, description={An open-source Python library providing scientific algorithms including sparse matrix solvers, signal processing, and optimisation routines}, }

\newglossaryentry{wakis}{
name={Wakis}, description={An open-source, GPU-accelerated 3D time-domain electromagnetic solver for the calculation of wake potential and beam-coupling impedance, developed at CERN during this thesis}, }

\newglossaryentry{warpx}{
name={WarpX}, description={An open-source, GPU-accelerated particle-in-cell code for electromagnetic plasma and beam-driven accelerator simulations}, }

\newglossaryentry{cst}{
name={CST}, description={A commercial electromagnetic simulation software suite widely used for beam-coupling impedance and wakefield calculations in particle accelerators}, }

\newglossaryentry{echo4d}{
name={ECHO4D}, description={A closed-source time-domain electromagnetic solver for short-range wakefield and impedance calculations, based on the Finite Integration Technique with a co-moving window}, }

\newglossaryentry{meep}{
name={Meep}, description={open-source electromagnetic simulation software based on the finite-difference time-domain method, developed for modelling wave propagation in complex geometries and materials}, }

\newglossaryentry{ace3p}{
name={ACE3P}, description={suite of parallel finite-element electromagnetic solvers developed at SLAC for the analysis of accelerator components in the frequency and time domains} }

\newglossaryentry{impedancewake2d}{
name={ImpedanceWake2D}, description={semi-analytical open-source software developed at CERN for calculating the beam-coupling impedance and wake functions of longitudinally uniform, multilayered beam pipes with circular or flat cross-sections. Available at https://gitlab.cern.ch/IRIS/IW2D}, , }


\newglossaryentry{embedded boundary}{
name={embedded boundary}, description={Geometrical surface that separates different material regions independently of the underlying computational grid} }

\newglossaryentry{boundary condition}{
name={boundary condition}, description={numerical condition imposed at the outer boundary of the computational domain to prescribe the behaviour of the electromagnetic fields and close the discrete system of equations } }

\newglossaryentry{partially_filled_cell}{
name={partially filled cell}, plural={partially filled cells}, description={Grid cell intersected by a material interface and therefore containing portions of two or more materials. It is also commonly called a cut cell} }

\newglossaryentry{staircase approximation}{
name={staircase approximation}, description={Method for representing a material interface in which every intersected cell is assigned entirely to one of the adjacent materials. As a result, curved or oblique boundaries are represented by grid-aligned steps} }


\newglossaryentry{subpixel smoothing}{
name={subpixel smoothing}, description={Method for representing a material interface by assigning effective material properties to the cells intersected by the boundary, based on the materials contained within each cell} }

\newglossaryentry{waveguide_mode}
{ name={waveguide mode}, description={electromagnetic field pattern that satisfies Maxwell's equations and the boundary conditions imposed by the cross-section of a waveguide} }

\newglossaryentry{cutoff_frequency}
{ name={cutoff frequency}, description={minimum frequency above which a given waveguide mode can propagate along the waveguide} }

\newglossaryentry{evanescent_field}
{ name={evanescent field}, description={non-propagating electromagnetic field whose amplitude decays exponentially in a given direction} }

\newglossaryentry{wave_impedance}
{ name={wave impedance}, description={ratio between the transverse electric- and magnetic-field amplitudes of an electromagnetic wave} }

\newglossaryentry{ghost_cell}
{ name={ghost cell}, description={additional grid cell outside the physical computational domain whose value is assigned to impose a boundary condition or exchange data between neighbouring computational subdomains} }

\newglossaryentry{adiabatic_absorber}
{ name={adiabatic absorbing layer}, description={lossy region in which the material absorption increases gradually so that outgoing waves are attenuated while reflections from the entrance are minimized} }

\newglossaryentry{wakelength}{
    name={wakelength},
    description={Maximum longitudinal separation between the source bunch and the trailing witness for which the wake potential is calculated in a time-domain simulation. It determines the simulated wake duration and influences the frequency resolution}
}
