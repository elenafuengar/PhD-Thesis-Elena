\newcommand{\chaptercolor}{%
  \ifnum\value{chapter}<3 cAqua\else

  cPink\fi\fi\fi
}

% --- mainmatter flag (front/back matter => black) ---
\newif\ifInMainmatter
\InMainmatterfalse

\let\origfrontmatter\frontmatter
\renewcommand{\frontmatter}{\origfrontmatter\InMainmatterfalse
  \hypersetup{linkcolor=black,citecolor=black,urlcolor=black}}

\let\origmainmatter\mainmatter
\renewcommand{\mainmatter}{\origmainmatter\InMainmattertrue}

\let\origbackmatter\backmatter
\renewcommand{\backmatter}{\origbackmatter\InMainmatterfalse
  \hypersetup{linkcolor=black,citecolor=black,urlcolor=black}}

% --- chapter color: only active in mainmatter ---
\newcommand{\chaptercolorof}[1]{%
  \ifInMainmatter
    \ifnum#1<3 cAqua\else
    \ifnum#1<5 cOrange\else
    \ifnum#1<7 cGreen\else
    cPink\fi\fi\fi
  \else
    black%
  \fi
}
\renewcommand{\chaptercolor}{\chaptercolorof{\value{chapter}}}
